"""Ownership and failure behavior of the in-memory PDF transfer resource."""
from multiprocessing.shared_memory import SharedMemory

import pytest

from app.payload import SharedPDF


def test_payload_contains_exact_bytes_and_releases_its_named_resource():
    body = b'%PDF-1.7\n\x00\xffbinary bytes'
    with SharedPDF.from_bytes(body) as payload:
        name = payload.name
        reader = SharedMemory(name=name)
        try:
            assert bytes(reader.buf[:payload.size]) == body
        finally:
            reader.close()
    with pytest.raises(FileNotFoundError):
        SharedMemory(name=name)
    payload.close()  # repeated disposal is harmless


def test_payload_is_released_if_its_consumer_fails():
    with pytest.raises(RuntimeError, match='consumer failed'):
        with SharedPDF.from_bytes(b'%PDF-test') as payload:
            name = payload.name
            raise RuntimeError('consumer failed')
    with pytest.raises(FileNotFoundError):
        SharedMemory(name=name)


def test_transfer_resource_exhaustion_returns_503_and_recovers(monkeypatch):
    import pymupdf
    from fastapi.testclient import TestClient
    from app.main import app
    import app.payload as payload_module

    with pymupdf.open() as document:
        document.new_page().insert_text((72, 72), 'Resource recovery')
        body = document.tobytes()

    def unavailable(*args, **kwargs):
        raise OSError('shared memory unavailable')

    with TestClient(app) as client:
        with monkeypatch.context() as patch:
            patch.setattr(payload_module, 'SharedMemory', unavailable)
            response = client.post('/extract', content=body, headers={'Content-Type': 'application/pdf'})
            assert response.status_code == 503
            assert client.get('/health').status_code == 200
        response = client.post('/extract', content=body, headers={'Content-Type': 'application/pdf'})
        assert response.status_code == 200
        assert response.json()['page_count'] == 1
        assert 'Resource recovery' in response.json()['content']
